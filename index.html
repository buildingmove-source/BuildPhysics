// ===== ЭКСПОРТ В РЕПОЗИТОРИЙ (через GitHub API) ===== const GITHUB_USERNAME = "buildingmove-source"; const GITHUB_REPO = "BuildPhysics"; let githubToken = null; // Загрузка токена из localStorage function loadGitHubToken() { const saved = localStorage.getItem('github_pat_token'); if (saved) githubToken = saved; } // Сохранение токена function saveGitHubToken(token) { githubToken = token; localStorage.setItem('github_pat_token', token); } // Показ панели ввода токена function showTokenPanel() { let panel = document.getElementById('tokenPanel'); if (!panel) { const adminPanel = document.querySelector('.admin-panel'); panel = document.createElement('div'); panel.id = 'tokenPanel'; panel.style.cssText = ` background: white; border: 1px solid #D0D7DE; border-radius: 8px; padding: 0.8rem; margin-bottom: 1rem; display: none; `; panel.innerHTML = `
🔑 Токен GitHub:
⚠️ Токен нужен для сохранения файла в репозиторий. Создать токен (нужны права: repo)
`; adminPanel.parentNode.insertBefore(panel, adminPanel.nextSibling); document.getElementById('saveTokenBtn')?.addEventListener('click', () => { const token = document.getElementById('githubTokenInput').value.trim(); if (token) { saveGitHubToken(token); panel.style.display = 'none'; showToast('🔑 Токен сохранён'); exportToRepository(); } else { showToast('❌ Введите корректный токен'); } }); } panel.style.display = 'block'; } async function exportToRepository() { loadGitHubToken(); if (!githubToken) { showTokenPanel(); showToast('⚠️ Введите токен GitHub для сохранения в репозиторий'); return; } const now = new Date(); const dateStr = `${now.getDate().toString().padStart(2,'0')}.${(now.getMonth()+1).toString().padStart(2,'0')}.${now.getFullYear()}`; const timeStr = `${now.getHours().toString().padStart(2,'0')}:${now.getMinutes().toString().padStart(2,'0')}`; const fileName = `Ресурсы_Огни3_${dateStr.replace(/\./g, '-')}.html`; // Формируем HTML для экспорта let exportRows = []; let currentSection = ''; resourcesData.forEach((item) => { if (currentSection !== item.section) { currentSection = item.section; exportRows.push({ type: 'section', section: item.section }); } const contractorsList = item.contractors.map(c => `${c.name}: ${c.people}`).join('; '); exportRows.push({ type: 'row', section: item.section, subsys: item.subsys, work: item.work, contractors: contractorsList, total: getRowTotal(item.contractors) }); }); const totalPeople = getTotalPeople(); const exportHtml = ` Ресурсы Огни 3 от ${dateStr}

Ресурсы · Трудовые ресурсы

АО «Огни 3» · Дата формирования: ${dateStr} ${timeStr}

${exportRows.map(row => { if (row.type === 'section') { return ``; } else { return ``; } }).join('')}
Объект / СекцияРаздел ВИСНаименование работПодрядная организация / кол-воВсего, чел.
${escapeHtml(row.section)}
${escapeHtml(row.section)}${escapeHtml(row.subsys)}${escapeHtml(row.work)}${escapeHtml(row.contractors)}${row.total}
ВСЕГО ЧЕЛОВЕК${totalPeople}
Данные актуальны на момент формирования · АО «Огни 3»
`; showToast('📤 Сохранение в репозиторий...'); try { // Проверяем, существует ли файл let sha = null; const getResponse = await fetch(`https://api.github.com/repos/${GITHUB_USERNAME}/${GITHUB_REPO}/contents/${fileName}`, { headers: { 'Authorization': `token ${githubToken}` } }); if (getResponse.ok) { const data = await getResponse.json(); sha = data.sha; } // Отправляем файл const uploadResponse = await fetch(`https://api.github.com/repos/${GITHUB_USERNAME}/${GITHUB_REPO}/contents/${fileName}`, { method: 'PUT', headers: { 'Authorization': `token ${githubToken}`, 'Content-Type': 'application/json' }, body: JSON.stringify({ message: `Экспорт ресурсов от ${dateStr} ${timeStr}`, content: btoa(unescape(encodeURIComponent(exportHtml))), sha: sha }) }); if (uploadResponse.ok) { showToast(`✅ Файл сохранён в репозиторий!`); // Обновляем список файлов на главной странице (через localStorage) localStorage.setItem('needRefreshFiles', 'true'); } else { const error = await uploadResponse.json(); showToast(`❌ Ошибка: ${error.message || 'Неизвестная ошибка'}`); } } catch(error) { showToast(`❌ Ошибка соединения: ${error.message}`); } } // Заменяем кнопку экспорта document.getElementById('exportBtn').onclick = exportToRepository;
